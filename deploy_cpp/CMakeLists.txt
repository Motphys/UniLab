cmake_minimum_required(VERSION 3.20)
project(unilab_rma_deploy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WITH_TENSORRT "Build TensorRT 10 policy inference" OFF)
option(ENABLE_TSAN "Enable thread sanitizer on supported Unix compilers" OFF)
find_package(Threads REQUIRED)
add_executable(control_loop src/main_control_loop.cpp)
add_executable(test_double_buffer tests/test_double_buffer.cpp)
add_executable(test_control_mapping tests/test_control_mapping.cpp)
foreach(target control_loop test_double_buffer test_control_mapping)
    target_include_directories(${target} PRIVATE include)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(ENABLE_TSAN)
            target_compile_options(${target} PRIVATE -fsanitize=thread -g)
            target_link_options(${target} PRIVATE -fsanitize=thread)
        endif()
    endif()
endforeach()
if(WITH_TENSORRT)
    find_package(CUDAToolkit REQUIRED)
    find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include REQUIRED)
    find_library(TENSORRT_LIBRARY nvinfer HINTS ${TENSORRT_ROOT}/lib ${TENSORRT_ROOT}/lib64 REQUIRED)
    target_compile_definitions(control_loop PRIVATE WITH_TENSORRT)
    target_include_directories(control_loop PRIVATE ${TENSORRT_INCLUDE})
    target_link_libraries(control_loop PRIVATE ${TENSORRT_LIBRARY} CUDA::cudart)
    add_executable(policy_worker src/policy_worker.cpp)
    target_include_directories(policy_worker PRIVATE include ${TENSORRT_INCLUDE})
    target_link_libraries(policy_worker PRIVATE ${TENSORRT_LIBRARY} CUDA::cudart)
    if(MSVC)
        target_compile_options(policy_worker PRIVATE /W4 /EHsc)
    else()
        target_compile_options(policy_worker PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
enable_testing()
add_test(NAME double_buffer_consistency COMMAND test_double_buffer)
add_test(NAME mock_control_loop COMMAND control_loop --seconds 1)
add_test(NAME policy_actuator_mapping COMMAND test_control_mapping)
